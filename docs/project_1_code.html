list_vars_21 <- function(api_url1){
  api_url1 <- 'https://api.census.gov/data/2021/acs/acs1/pums/variables.json'
  data_url1 <- httr::GET(api_url1)
  parsed_data1 <- fromJSON(rawToChar(data_url1$content))
}
variables_2021 <- list_vars_21(api_url1)
