#Writing function that returns numeric variables, also changes them from characters to numeric
return_num_vars <- function(year, target_var = 'AGEP'){
  rds <- paste0(year, "_variables.rds")
  var_list <- readRDS(rds)
  PWGTP = c(var_list$variables$PWGTP$values$range$min, var_list$variables$PWGTP$values$range$max)
  AGEP = c(var_list$variables$AGEP$values$range$min, var_list$variables$AGEP$values$range$max)
  GASP = c(var_list$variables$GASP$values$range$min, var_list$variables$GASP$values$range$max)
  GRPIP = c(var_list$variables$GRPIP$values$range$min, var_list$variables$GRPIP$values$range$max)
  JWMNP = c(var_list$variables$JWMNP$values$range$min, var_list$variables$JWMNP$values$range$max)
  full_list <- list(AGEP = AGEP, GASP = GASP, GRPIP = GRPIP, JWMNP = JWMNP, PWGTP = PWGTP)
  num_list <- lapply(full_list, as.numeric, na.rm = TRUE)
  return(list(num_list['PWGTP'], num_list[target_var]))
}

to_minutes <- function(time_str, am_pm){
  parts <- str_split(time_str, ':', simplify = TRUE)
  hours <- as.numeric(parts[, 1])
  minutes <- as.numeric(parts[, 2])
  
  is_pm <- str_detect(am_pm, 'p')
  
  hours <- case_when(
    is_pm & hours != 12 ~ hours + 12,
    !is_pm & hours == 12 ~ 0,

  )
  return(hours * 60 + minutes)
}

summary.census <- function(year, numeric_var, cat_var){
  rds <- paste0(year, '_variables.rds')
  var_list <- readRDS(rds)
  
  PWGTP = c(var_list$variables$PWGTP$values$range$min, var_list$variables$PWGTP$values$range$max)
  AGEP = c(var_list$variables$AGEP$values$range$min, var_list$variables$AGEP$values$range$max)
  GASP = c(var_list$variables$GASP$values$range$min, var_list$variables$GASP$values$range$max)
  GRPIP = c(var_list$variables$GRPIP$values$range$min, var_list$variables$GRPIP$values$range$max)
  JWMNP = c(var_list$variables$JWMNP$values$range$min, var_list$variables$JWMNP$values$range$max)
  full_list <- list(PWGTP = PWGTP, AGEP = AGEP, GASP = GASP, GRPIP = GRPIP, JWMNP = JWMNP)
  num_list <- lapply(full_list, as.numeric, na.rm = TRUE)
  numeric_var <- num_list[[numeric_var]]
  weight_vector <- num_list[['PWGTP']]
  sample_mean <- sum(numeric_var*weight_vector)/sum(weight_vector)
  sample_std_dev <- sqrt(sum(numeric_var*numeric_var*weight_vector)/sum(weight_vector)-sample_mean*sample_mean)
  
  FER = factor(unlist(var_list$variables$FER$values$item))
  HHL = factor(unlist(var_list$variables$HHL$values$item))
  SCH = factor(unlist(var_list$variables$SCH$values$item))
  SCHL = factor(unlist(var_list$variables$SCHL$values$item))
  SEX = factor(unlist(var_list$variables$SEX$values$item))
  var_list <- list(FER = FER, HHL = HHL, SCH = SCH, SCHL = SCHL, SEX = SEX)
  cat_var <- var_list[cat_var]
  return(list(sample_mean, sample_std_dev, cat_var))
}
